cmake_minimum_required(VERSION 3.16)
project(step3_vascular_pure_fluid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(MPI REQUIRED COMPONENTS CXX)
get_filename_component(HC "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(HEMOCELL_ARCHIVE "${HC}/build/libhemocell.a" CACHE FILEPATH "Existing verified archive, read only")
if(NOT EXISTS "${HEMOCELL_ARCHIVE}")
  message(FATAL_ERROR "Existing HemoCell archive absent")
endif()
add_executable(vascular_pure_fluid vascularPureFluid.cpp)
target_include_directories(vascular_pure_fluid PRIVATE "${HC}/palabos/src" "${HC}/palabos/externalLibraries")
target_compile_definitions(vascular_pure_fluid PRIVATE PLB_MPI_PARALLEL PLB_SMP_PARALLEL PLB_USE_POSIX)
target_compile_options(vascular_pure_fluid PRIVATE -O2 -Wno-deprecated-declarations)
target_link_libraries(vascular_pure_fluid PRIVATE "${HEMOCELL_ARCHIVE}" MPI::MPI_CXX)
