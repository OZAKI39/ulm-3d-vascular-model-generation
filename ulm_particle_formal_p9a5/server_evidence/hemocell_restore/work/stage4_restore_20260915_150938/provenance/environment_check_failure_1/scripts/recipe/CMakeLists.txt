cmake_minimum_required(VERSION 3.16)
project(Vast4090Stage4Restore LANGUAGES C CXX)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(TBB REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT RESTORE_GPU_ARCH MATCHES "^[0-9]+$")
  message(FATAL_ERROR "RESTORE_GPU_ARCH must come from actual allocated GPU compute_cap")
endif()
get_filename_component(RESTORE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PALABOS_ROOT "${RESTORE_ROOT}/palabos")
add_library(native_palabos STATIC IMPORTED)
set_target_properties(native_palabos PROPERTIES IMPORTED_LOCATION "${RESTORE_GPU_BUILD_ROOT}/official/libpalabos.a")
add_executable(vascularPoC "${RESTORE_ROOT}/source/vascularPoC.cpp")
target_include_directories(vascularPoC PRIVATE "${PALABOS_ROOT}/src" "${PALABOS_ROOT}/externalLibraries" "${PALABOS_ROOT}/externalLibraries/Eigen3")
target_compile_definitions(vascularPoC PRIVATE PLB_MPI_PARALLEL PLB_USE_POSIX GPU_POC USE_CUDA_MALLOC OMPI_SKIP_MPICXX)
target_compile_options(vascularPoC PRIVATE -stdpar -Kieee -gpu=cc${RESTORE_GPU_ARCH},nofma -fopenmp -O3)
target_link_options(vascularPoC PRIVATE -stdpar -Kieee -gpu=cc${RESTORE_GPU_ARCH},nofma -fopenmp)
target_link_libraries(vascularPoC PRIVATE native_palabos MPI::MPI_CXX TBB::tbb)
