cmake_minimum_required(VERSION 3.16)
project(NewMediumSmoke LANGUAGES C CXX)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(TBB REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SMOKE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BASELINE_ROOT "/workspace/hemocell_restore/work/stage4_restore_20260915_150938")
set(PALABOS_ROOT "${BASELINE_ROOT}/palabos")
add_library(native_palabos STATIC IMPORTED)
set_target_properties(native_palabos PROPERTIES IMPORTED_LOCATION "${BASELINE_ROOT}/build/official/libpalabos.a")
add_executable(vascularPoC "${SMOKE_ROOT}/source/vascularPoC.cpp")
target_include_directories(vascularPoC PRIVATE "${PALABOS_ROOT}/src" "${PALABOS_ROOT}/externalLibraries" "${PALABOS_ROOT}/externalLibraries/Eigen3")
target_compile_definitions(vascularPoC PRIVATE PLB_MPI_PARALLEL PLB_USE_POSIX GPU_POC USE_CUDA_MALLOC OMPI_SKIP_MPICXX)
target_compile_options(vascularPoC PRIVATE -stdpar -Kieee -gpu=cc89,nofma -fopenmp -O3)
target_link_options(vascularPoC PRIVATE -stdpar -Kieee -gpu=cc89,nofma -fopenmp)
target_link_libraries(vascularPoC PRIVATE native_palabos MPI::MPI_CXX TBB::tbb)
