cmake_minimum_required(VERSION 3.16)
project(step2_vascular_voxelization_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(MPI REQUIRED COMPONENTS CXX)
get_filename_component(HC "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(HEMOCELL_ARCHIVE "${HC}/build/libhemocell.a" CACHE FILEPATH "Existing verified static archive; read only")
if(NOT EXISTS "${HEMOCELL_ARCHIVE}")
  message(FATAL_ERROR "STEP2_BUILD_INTEGRATION=BLOCKED: existing archive absent")
endif()
add_executable(closed_voxelizer closedVascularVoxelizer.cpp)
target_include_directories(closed_voxelizer PRIVATE "${HC}/palabos/src" "${HC}/palabos/externalLibraries")
target_compile_definitions(closed_voxelizer PRIVATE PLB_MPI_PARALLEL PLB_SMP_PARALLEL PLB_USE_POSIX)
target_compile_options(closed_voxelizer PRIVATE -O2 -Wno-deprecated-declarations)
target_link_libraries(closed_voxelizer PRIVATE "${HEMOCELL_ARCHIVE}" MPI::MPI_CXX)
