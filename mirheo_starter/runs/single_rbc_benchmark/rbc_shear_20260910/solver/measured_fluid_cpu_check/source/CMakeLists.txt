cmake_minimum_required(VERSION 3.10)
project(single_rbc_shear_benchmark LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
get_filename_component(HEMO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(HEMO_SOURCE "${HEMO_ROOT}/vendor/HemoCell")
find_package(MPI REQUIRED)
set(HDF5_PREFER_PARALLEL ON)
find_package(HDF5 REQUIRED COMPONENTS C HL)
add_executable(single_rbc_shear_benchmark benchmark.cpp)
target_compile_definitions(single_rbc_shear_benchmark PRIVATE PLB_MPI_PARALLEL PLB_SMP_PARALLEL PLB_USE_POSIX)
target_compile_options(single_rbc_shear_benchmark PRIVATE -O3 -march=native -Wno-deprecated-declarations)
foreach(dir config core helper io mechanics external/tinyxml2 palabos/src palabos/externalLibraries/tinyxml palabos/externalLibraries .)
  target_include_directories(single_rbc_shear_benchmark PRIVATE "${HEMO_SOURCE}/${dir}")
endforeach()
target_include_directories(single_rbc_shear_benchmark PRIVATE ${HDF5_INCLUDE_DIRS})
target_link_libraries(single_rbc_shear_benchmark PRIVATE "${HEMO_ROOT}/build/upstream/libhemocell.a" MPI::MPI_CXX ${HDF5_LIBRARIES} ${HDF5_HL_LIBRARIES})
