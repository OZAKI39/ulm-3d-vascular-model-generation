cmake_minimum_required(VERSION 3.16)
project(VascularPoC LANGUAGES C CXX)
find_package(MPI REQUIRED COMPONENTS CXX)
if(GPU_POC)
  set(PALABOS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../upstream/palabos")
  set(PALABOS_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../build_gpu/official/libpalabos.a")
  set(CMAKE_CXX_STANDARD 20)
  find_package(TBB REQUIRED)
else()
  set(PALABOS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../frozen_bundle/source/native/palabos")
  set(PALABOS_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../build_cpu/libstep3b_native.a")
  set(CMAKE_CXX_STANDARD 11)
  find_package(HDF5 REQUIRED COMPONENTS C)
endif()
add_library(native_palabos STATIC IMPORTED)
set_target_properties(native_palabos PROPERTIES IMPORTED_LOCATION "${PALABOS_LIBRARY}")
add_executable(vascularPoC vascularPoC.cpp)
target_include_directories(vascularPoC PRIVATE "${PALABOS_ROOT}/src" "${PALABOS_ROOT}/externalLibraries" "${PALABOS_ROOT}/externalLibraries/Eigen3")
target_compile_definitions(vascularPoC PRIVATE PLB_MPI_PARALLEL PLB_USE_POSIX)
target_link_libraries(vascularPoC PRIVATE native_palabos MPI::MPI_CXX)
if(GPU_POC)
  target_compile_definitions(vascularPoC PRIVATE GPU_POC USE_CUDA_MALLOC OMPI_SKIP_MPICXX)
  target_compile_options(vascularPoC PRIVATE -stdpar -Kieee -gpu=cc120,nofma -fopenmp -O3)
  target_link_options(vascularPoC PRIVATE -stdpar -Kieee -gpu=cc120,nofma -fopenmp)
  target_link_libraries(vascularPoC PRIVATE TBB::tbb)
else()
  target_compile_definitions(vascularPoC PRIVATE PLB_SMP_PARALLEL)
  target_compile_options(vascularPoC PRIVATE -O2 -Wno-deprecated)
  target_include_directories(vascularPoC PRIVATE ${HDF5_INCLUDE_DIRS})
  target_link_libraries(vascularPoC PRIVATE ${HDF5_LIBRARIES})
endif()
