cmake_minimum_required(VERSION 3.16)
project(rbc_stage1_cpu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(MPI REQUIRED COMPONENTS C CXX)
set(HDF5_PREFER_PARALLEL TRUE)
find_package(HDF5 REQUIRED COMPONENTS C HL)
if(NOT HDF5_IS_PARALLEL)
 message(FATAL_ERROR "Parallel HDF5 required")
endif()
set(N ${CMAKE_CURRENT_SOURCE_DIR}/native)
file(GLOB_RECURSE FILES CONFIGURE_DEPENDS ${N}/palabos/src/*.cpp ${N}/palabos/externalLibraries/tinyxml/*.cpp)
foreach(PART config core helper io mechanics external/tinyxml2)
 file(GLOB_RECURSE F CONFIGURE_DEPENDS ${N}/${PART}/*.cpp)
 list(APPEND FILES ${F})
endforeach()
add_library(hemocell_cpu STATIC ${FILES})
target_include_directories(hemocell_cpu PUBLIC ${N} ${N}/palabos/src ${N}/palabos/externalLibraries ${N}/palabos/externalLibraries/tinyxml ${N}/config ${N}/core ${N}/helper ${N}/io ${N}/mechanics ${N}/external/tinyxml2 ${HDF5_INCLUDE_DIRS})
target_compile_definitions(hemocell_cpu PUBLIC PLB_MPI_PARALLEL PLB_SMP_PARALLEL PLB_USE_POSIX)
target_compile_definitions(hemocell_cpu PRIVATE COMPILING_HEMOCELL_LIBRARY)
target_compile_options(hemocell_cpu PRIVATE -O3 -Wno-deprecated-declarations)
target_link_libraries(hemocell_cpu PUBLIC MPI::MPI_CXX ${HDF5_LIBRARIES} ${HDF5_HL_LIBRARIES})
add_executable(rbc_geometry_probe rbc_geometry_probe.cpp)
target_link_libraries(rbc_geometry_probe PRIVATE hemocell_cpu)
target_compile_options(rbc_geometry_probe PRIVATE -O2 -Wno-deprecated-declarations)
