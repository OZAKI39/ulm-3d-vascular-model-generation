"""Mechanical extraction of the existing runner; no dynamic loading at runtime."""
from pathlib import Path
import json, difflib
from inventory_sources import ROOT, B, sha, dump

def main():
    source=B/'scripts/solve_remote.py';text=source.read_text()
    text=text[:text.index("if __name__=='__main__':")]
    start=text.index('parser=argparse.ArgumentParser()');end=text.index('from vascular_validation.postprocess')
    text=text[:start]+text[end:]
    text=text.replace('def main():\n    case = args.case; run = case/\'run\'; reports = case/\'reports\'',
                      "def run(case, native, measurement_class=SolutionMeasurements):\n    case=Path(case).resolve(); run=case/'run'; reports=case/'reports'")
    start=text.index("    stack=Path('/workspace/");end=text.index("    options=(run/'PETSC_OPTIONS.txt')",start)
    text=text[:start]+'''    exe=native['executable']
    if not exe:raise ValueError('Native production binary is not configured')
    assert sha(exe)==native['executable_sha256']
    library=Path(native['petsc_library'])
    assert sha(library)==native['petsc_library_sha256']
    w=dict(candidate_wrapper=native['wrapper'],launcher=native['mpi_launcher'])
    build=dict(runtime_library_path=native['runtime_library_path'])
'''+text[end:]
    text=text.replace('remote=True','remote=False')
    text=text.replace("measure=SolutionMeasurements(case/'SV_MESH/mesh_arrays.npz'", "measure=measurement_class(Path(policy['mesh_arrays'])")
    text=text.replace("sha(Path(w['prefix'])/'lib/libpetsc.so')",'sha(library)')
    text=text.replace("PATH='/usr/bin:/bin:/usr/local/cuda/bin'", "PATH='/usr/bin:/bin:/usr/local/cuda-12.6/bin:/usr/lib/wsl/lib'")
    text=text.replace('    print(json.dumps({k:record[k]', '    print(json.dumps({k:record[k]')
    text += '\n    return record\n'
    target=ROOT/'src/human_vessel_flow/native_runner.py';target.write_text(text)
    dump(ROOT/'docs/provenance/NATIVE_RUNNER_REFACTOR.json',dict(source=str(source),source_sha256=sha(source),
        destination=str(target.relative_to(ROOT)),new_sha256=sha(target),changes=['Explicit case and native config arguments','Normal package imports','Explicit canonical mesh path','Local host metadata','No numerical gates or PETSc options changed']))
    (ROOT/'docs/provenance/NATIVE_RUNNER_REFACTOR.patch').write_text(''.join(difflib.unified_diff(source.read_text().splitlines(True),text.splitlines(True),fromfile=str(source),tofile=str(target))))

if __name__=='__main__':main()
