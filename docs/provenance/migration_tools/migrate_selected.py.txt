"""One-time explicit selection. Sources are read-only; a hashed plan precedes copying."""
from pathlib import Path
import hashlib, json, shutil, os, collections
from inventory_sources import ROOT, PROJECTS, B, P, SNAP, sha, dump

FILES = {}
OMITTED = []

def add(source, target, kind='immutable_data'):
    source = Path(source)
    if not source.is_file() or source.is_symlink():
        raise ValueError(('Expected real source file', source))
    target = str(target)
    if target in FILES and FILES[target]['source'] != str(source):
        raise ValueError(('Destination collision', target))
    FILES[target] = dict(source=str(source), destination=target, category=kind,
                         size=source.stat().st_size, sha256=sha(source))

def tree(source, target, kind='immutable_data', choose=lambda p: True):
    source = Path(source)
    for parent, dirs, names in os.walk(source):
        dirs[:] = sorted(d for d in dirs if d not in {'.git','__pycache__','.pytest_cache','.venv','env','build'} and not (Path(parent)/d).is_symlink())
        for name in sorted(names):
            p=Path(parent)/name
            if p.is_symlink(): continue
            rel=p.relative_to(source)
            if p.suffix in {'.pyc','.so','.o'} or not choose(rel):
                OMITTED.append(dict(source=str(p),size=p.stat().st_size,reason='Selected scope excludes generated cache, duplicate frame or superseded execution'))
                continue
            add(p,Path(target)/rel,kind)

def main():
    import argparse
    ap=argparse.ArgumentParser();ap.add_argument('--copy',action='store_true');args=ap.parse_args()
    path=ROOT/'docs/provenance/SOURCE_MANIFEST_BEFORE_COPY.json'
    if path.exists():
        manifest=json.loads(path.read_text())
    else:
        plan();manifest=list(FILES.values());dump(path,manifest)
        dump(ROOT/'docs/provenance/SELECTION_OMISSIONS.json',OMITTED)
    print(json.dumps(dict(files=len(manifest),bytes=sum(r['size'] for r in manifest),categories=dict(collections.Counter(r['category'] for r in manifest)))))
    if not args.copy:return
    receipt=[]
    for r in manifest:
        src=Path(r['source']);dst=ROOT/r['destination']
        if sha(src)!=r['sha256']:raise ValueError(('Source changed after planning',src))
        if dst.exists():
            if sha(dst)!=r['sha256']:raise ValueError(('Refuse overwrite',dst))
        else:
            dst.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(src,dst)
        actual=sha(dst)
        if actual!=r['sha256']:raise ValueError(('Copy failed',dst))
        receipt.append(dict(destination=r['destination'],sha256=actual,status='PASS'))
    dump(ROOT/'docs/provenance/COPY_VERIFIED.json',receipt)
    print('PASS: every selected destination matches its pre-copy source hash')

if __name__=='__main__':main()
