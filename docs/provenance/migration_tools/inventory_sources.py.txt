"""Read-only inventory of the pre-migration workspace; never imports old code."""
from pathlib import Path
import ast, collections, hashlib, json, os, subprocess

ROOT = Path(__file__).resolve().parents[1]
PROJECTS = ROOT.parent
B = PROJECTS / 'computation_examples/brava_flow_roi_18mlmin'
P = PROJECTS / 'computation_examples/_shared/particle_workflow'
SNAP = PROJECTS / 'temp_storage/brava_flow_github_sync_20261022'

def sha(p):
    with p.open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()

def dump(p, value):
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

def git_record(root):
    def git(*args):
        return subprocess.run(['git', '-C', str(root), *args], capture_output=True, text=True).stdout
    rows = git('status', '--porcelain=v1', '--untracked-files=all').splitlines()
    code = [r for r in rows if r.endswith(('.py', '.cpp', '.cu', '.h', '.f90', '.json', '.xml')) and not r.startswith(' D')]
    return dict(root=str(root), head=git('rev-parse', 'HEAD').strip(), worktrees=git('worktree', 'list', '--porcelain'),
                status_counts=dict(collections.Counter(r[:2] for r in rows)), nondeleted_code_changes=code)

def particle_graph():
    src = P / 'particle_3d/src'
    modules = {'.'.join(p.relative_to(src).with_suffix('').parts).removesuffix('.__init__'): p
               for p in src.rglob('*.py')}
    graph = {}
    for name, path in modules.items():
        package = name if path.name == '__init__.py' else name.rsplit('.', 1)[0]
        imports = set()
        for node in ast.walk(ast.parse(path.read_text())):
            if isinstance(node, ast.Import):
                imports.update(a.name for a in node.names)
            elif isinstance(node, ast.ImportFrom):
                prefix = ('.'.join(package.split('.')[:len(package.split('.'))-node.level+1]) + '.') if node.level else ''
                module = (prefix + (node.module or '')).rstrip('.')
                imports.add(module)
                imports.update(module + '.' + a.name for a in node.names)
        graph[name] = sorted(imports & modules.keys())
    roots = {n for n in modules if n.startswith('particle_3d.population_v1')}
    chosen = set(roots)
    while True:
        new = set().union(*(set(graph[n]) for n in chosen)) - chosen
        if not new: break
        chosen |= new
    chosen.add('particle_3d')
    return dict(roots=sorted(roots), modules=sorted(chosen), edges=graph,
                source_paths={n:str(modules[n]) for n in sorted(chosen)},
                excluded_modules=sorted(modules.keys()-chosen))

def main():
    out = ROOT / 'docs/provenance'
    out.mkdir(parents=True, exist_ok=True)
    graph = particle_graph()
    dump(out/'PARTICLE_DEPENDENCY_GRAPH.json', graph)
    records = [git_record(r) for r in (B, P, SNAP, PROJECTS/'computation_examples/_archives/github_sync_20260928_latest')]
    dump(out/'GIT_WORKTREES_BEFORE.json', records)
    included = json.loads((SNAP/'sync_20261022/INCLUDED.json').read_text())
    comparison = []
    for r in included:
        source = Path(r['source'])
        if not str(source).startswith('/home/'): continue
        exists = source.is_file()
        value = sha(source) if exists else None
        comparison.append(dict(source=str(source), snapshot_path=r['path'], snapshot_sha256=r['sha256'],
                               local_sha256=value, status='SAME' if value==r['sha256'] else ('DIFFERENT' if exists else 'ABSENT')))
    dump(out/'LOCAL_GITHUB_COMPARISON.json', comparison)
    excluded = json.loads((SNAP/'sync_20261022/EXCLUDED.json').read_text())
    big = [r for r in excluded if r.get('size',0)>=100*2**20 and str(r.get('source','')).startswith('/home/')]
    dump(out/'GITHUB_UNPUBLISHED_LARGE_REVIEW.json', big)
    links=[]
    for base in (B, P, PROJECTS/'vascular_printing'):
        for parent, dirs, files in os.walk(base):
            dirs[:] = [d for d in dirs if d not in {'.git','.venv','__pycache__','vessel_model','Ultraliser'}]
            for name in dirs+files:
                p=Path(parent)/name
                if p.is_symlink(): links.append(dict(path=str(p),target=os.readlink(p),resolved=str(p.resolve()),exists=p.exists()))
    dump(out/'SOURCE_SYMLINKS.json',links)
    summary=dict(particle_runtime_modules=len(graph['modules']),github_comparison=dict(collections.Counter(r['status'] for r in comparison)),
                 unpublished_large_files=len(big), unpublished_bytes=sum(r['size'] for r in big),source_links=len(links),
                 git=[dict(root=r['root'],head=r['head'],status_counts=r['status_counts'],nondeleted_code_changes=len(r['nondeleted_code_changes'])) for r in records])
    dump(out/'INVENTORY_SUMMARY.json',summary)
    print(json.dumps(summary,ensure_ascii=False,indent=2))

if __name__ == '__main__': main()
