"""Expose original P1 acceptance/WSS recovery for a new case, never ACTIVE_FLOW."""
from pathlib import Path
import difflib
from inventory_sources import ROOT,B,sha,dump

def main():
    source=B/'scripts/postprocess_flow.py';text=source.read_text()
    text=text.replace("ROOT=Path(__file__).resolve().parents[1];sys.path.insert(0,str(ROOT/'vendor'))",'from .paths import ROOT\nimport xml.etree.ElementTree as ET')
    start=text.index('def main():');end=text.index(" e=json.loads",start)
    text=text[:start]+'''def postprocess(case):
 case=Path(case).resolve()
 if case.is_relative_to(ROOT/'cases') or (case/'frozen_flow').exists() or (case/'postprocess').exists():
  raise ValueError('Only a new, unprocessed case may be frozen; historical results are protected')
 xml=ET.parse(case/'run/solver.xml')
 mesh_root=Path(xml.findtext('.//Mesh_file_path')).resolve().parent
'''+text[end:]
    text=text.replace("case/'SV_MESH/mesh-complete.mesh.vtu'","mesh_root/'mesh-complete.mesh.vtu'")
    text=text.replace("case/'SV_MESH/mesh_arrays.npz'","mesh_root/'mesh_arrays.npz'")
    text=text.replace("case/'SV_MESH/mesh-complete.exterior.vtp'","mesh_root/'mesh-complete.exterior.vtp'")
    text=text.replace("  dump(ROOT/'reports/ACTIVE_FLOW.json',{**record,'case':str(case.relative_to(ROOT))})", "  # A new execution never replaces the historical active BraVa flow.\n  dump(case/'reports/NEW_FLOW_IDENTITY.json',{**record,'local_accuracy_certification':'NOT_RUN','historical_active_flow_replaced':False})")
    text=text.replace("if __name__=='__main__':main()",' return record')
    target=ROOT/'src/human_vessel_flow/postprocess.py';target.write_text(text)
    dump(ROOT/'docs/provenance/POSTPROCESS_REFACTOR.json',dict(source=str(source),source_sha256=sha(source),destination=str(target.relative_to(ROOT)),new_sha256=sha(target),changes=['Explicit new case','Explicit canonical mesh path from effective XML','Immutable result guards','New identity kept separate from historical ACTIVE_FLOW; original numerical gates retained']))
    (ROOT/'docs/provenance/POSTPROCESS_REFACTOR.patch').write_text(''.join(difflib.unified_diff(source.read_text().splitlines(True),text.splitlines(True),fromfile=str(source),tofile=str(target))))

if __name__=='__main__':main()
