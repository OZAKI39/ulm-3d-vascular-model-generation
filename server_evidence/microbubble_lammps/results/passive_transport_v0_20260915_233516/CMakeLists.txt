cmake_minimum_required(VERSION 3.18)
project(PassiveTransportV0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(MPI REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HDF5 REQUIRED IMPORTED_TARGET hdf5)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(ZLIB REQUIRED)
set(ENGINE_ROOT "" CACHE PATH "Read-only validated LAMMPS work root")
add_library(frozen_flow_coupling SHARED src/frozen_flow.cpp src/wall_distance.cpp)
target_include_directories(frozen_flow_coupling PUBLIC src)
target_link_libraries(frozen_flow_coupling PUBLIC PkgConfig::HDF5)
add_executable(wall_audit src/wall_audit.cpp)
target_link_libraries(wall_audit PRIVATE frozen_flow_coupling)
foreach(backend IN ITEMS cpu gpu)
 add_executable(passive_lmp_${backend} src/passive_main.cpp src/fix_passive_flow.cpp)
 target_include_directories(passive_lmp_${backend} PRIVATE ${ENGINE_ROOT}/upstream/lammps/src)
 target_compile_definitions(passive_lmp_${backend} PRIVATE LAMMPS_SMALLBIG)
 target_link_libraries(passive_lmp_${backend} PRIVATE ${ENGINE_ROOT}/build_${backend}/liblammps.a frozen_flow_coupling MPI::MPI_CXX JPEG::JPEG PNG::PNG ZLIB::ZLIB m dl)
 if(backend STREQUAL "gpu")
  target_link_libraries(passive_lmp_${backend} PRIVATE ${ENGINE_ROOT}/build_gpu/lib/kokkos/containers/src/libkokkoscontainers.a ${ENGINE_ROOT}/build_gpu/lib/kokkos/core/src/libkokkoscore.a /usr/local/cuda/lib64/stubs/libcuda.so /usr/local/cuda-13.2/targets/x86_64-linux/lib/libcudart.so ${ENGINE_ROOT}/build_gpu/lib/kokkos/simd/src/libkokkossimd.a)
 endif()
endforeach()
