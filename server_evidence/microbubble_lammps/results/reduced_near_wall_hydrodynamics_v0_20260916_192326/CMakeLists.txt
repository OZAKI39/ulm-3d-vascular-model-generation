cmake_minimum_required(VERSION 3.16)
project(reduced_wall_phase1r LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(HDF5 REQUIRED COMPONENTS C)
find_package(MPI REQUIRED COMPONENTS C)
add_library(reduced_wall_kernel src/reduced_wall_kernel.cpp)
target_include_directories(reduced_wall_kernel PUBLIC src ${HDF5_INCLUDE_DIRS})
target_link_libraries(reduced_wall_kernel PUBLIC ${HDF5_LIBRARIES} MPI::MPI_C)
target_compile_options(reduced_wall_kernel PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(test_reduced_wall tests/test_reduced_wall.cpp)
target_link_libraries(test_reduced_wall PRIVATE reduced_wall_kernel)
target_compile_options(test_reduced_wall PRIVATE -Wall -Wextra -Wpedantic -Werror)
