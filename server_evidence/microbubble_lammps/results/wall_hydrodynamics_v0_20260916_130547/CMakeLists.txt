cmake_minimum_required(VERSION 3.18)
project(WallHydrodynamicsV0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(MPI REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HDF5 REQUIRED IMPORTED_TARGET hdf5)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(ZLIB REQUIRED)
find_path(EIGEN_INCLUDE_DIR Eigen/Dense PATH_SUFFIXES eigen3 REQUIRED)
set(ENGINE_ROOT "" CACHE PATH "Read-only frozen LAMMPS 22Jul2025 Update6 root")
add_library(wall_math SHARED src/rigid_math.cpp src/frozen_flow.cpp src/wall_distance.cpp src/wall_model.cpp)
target_include_directories(wall_math PUBLIC src PRIVATE ${EIGEN_INCLUDE_DIR})
target_link_libraries(wall_math PUBLIC PkgConfig::HDF5)
foreach(backend IN ITEMS cpu gpu)
 add_executable(wall_lmp_${backend} src/rigid_lammps_main.cpp)
 target_include_directories(wall_lmp_${backend} PRIVATE ${ENGINE_ROOT}/upstream/lammps/src)
 target_compile_definitions(wall_lmp_${backend} PRIVATE LAMMPS_SMALLBIG)
 target_link_libraries(wall_lmp_${backend} PRIVATE ${ENGINE_ROOT}/build_${backend}/liblammps.a wall_math MPI::MPI_CXX JPEG::JPEG PNG::PNG ZLIB::ZLIB m dl)
 if(backend STREQUAL "gpu")
  target_link_libraries(wall_lmp_${backend} PRIVATE ${ENGINE_ROOT}/build_gpu/lib/kokkos/containers/src/libkokkoscontainers.a ${ENGINE_ROOT}/build_gpu/lib/kokkos/core/src/libkokkoscore.a /usr/local/cuda/lib64/stubs/libcuda.so /usr/local/cuda-13.2/targets/x86_64-linux/lib/libcudart.so ${ENGINE_ROOT}/build_gpu/lib/kokkos/simd/src/libkokkossimd.a)
 endif()
endforeach()
