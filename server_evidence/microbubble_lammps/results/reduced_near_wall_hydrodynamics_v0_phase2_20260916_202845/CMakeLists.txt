cmake_minimum_required(VERSION 3.18)
project(ReducedNearWallPhase2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(MPI REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HDF5 REQUIRED IMPORTED_TARGET hdf5)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(ZLIB REQUIRED)
set(ENGINE_ROOT "" CACHE PATH "Frozen stable LAMMPS 22Jul2025 Update6")
add_library(rigid_math SHARED src/rigid_math.cpp src/frozen_flow.cpp src/reduced_wall_kernel.cpp src/flat_wall_engine.cpp)
target_include_directories(rigid_math PUBLIC src)
target_link_libraries(rigid_math PUBLIC PkgConfig::HDF5 MPI::MPI_CXX)
target_compile_options(rigid_math PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rigid_lmp_cpu src/rigid_lammps_main.cpp)
target_include_directories(rigid_lmp_cpu PRIVATE ${ENGINE_ROOT}/upstream/lammps/src)
target_compile_definitions(rigid_lmp_cpu PRIVATE LAMMPS_SMALLBIG)
target_link_libraries(rigid_lmp_cpu PRIVATE ${ENGINE_ROOT}/build_cpu/liblammps.a rigid_math MPI::MPI_CXX JPEG::JPEG PNG::PNG ZLIB::ZLIB m dl)
target_compile_options(rigid_lmp_cpu PRIVATE -Wall -Wextra -Wpedantic)
add_executable(test_phase2_contract tests/test_phase2_contract.cpp)
target_link_libraries(test_phase2_contract PRIVATE rigid_math)
target_compile_options(test_phase2_contract PRIVATE -Wall -Wextra -Wpedantic -Werror)
