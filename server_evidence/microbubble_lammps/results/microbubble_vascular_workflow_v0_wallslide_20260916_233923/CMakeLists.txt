cmake_minimum_required(VERSION 3.18)
project(WallSlidingWorkflowV0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(MPI REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HDF5 REQUIRED IMPORTED_TARGET hdf5)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(ZLIB REQUIRED)
set(ENGINE_ROOT "" CACHE PATH "Frozen LAMMPS 22Jul2025 Update6 root")
add_library(rigid_math SHARED src/rigid_math.cpp src/frozen_flow.cpp src/wall_distance.cpp src/kinematic_wall_constraint.cpp)
target_include_directories(rigid_math PUBLIC src)
target_link_libraries(rigid_math PUBLIC PkgConfig::HDF5)
add_executable(workflow_lmp src/workflow_main.cpp)
target_include_directories(workflow_lmp PRIVATE ${ENGINE_ROOT}/upstream/lammps/src)
target_compile_definitions(workflow_lmp PRIVATE LAMMPS_SMALLBIG)
target_link_libraries(workflow_lmp PRIVATE ${ENGINE_ROOT}/build_cpu/liblammps.a rigid_math MPI::MPI_CXX JPEG::JPEG PNG::PNG ZLIB::ZLIB m dl)
add_executable(test_injection src/test_injection.cpp)
target_compile_options(test_injection PRIVATE -UNDEBUG)
add_executable(test_ports_lifecycle src/test_ports_lifecycle.cpp)
target_include_directories(test_ports_lifecycle PRIVATE ${ENGINE_ROOT}/upstream/lammps/src)
target_compile_definitions(test_ports_lifecycle PRIVATE LAMMPS_SMALLBIG)
target_compile_options(test_ports_lifecycle PRIVATE -UNDEBUG)
target_link_libraries(test_ports_lifecycle PRIVATE ${ENGINE_ROOT}/build_cpu/liblammps.a rigid_math MPI::MPI_CXX JPEG::JPEG PNG::PNG ZLIB::ZLIB m dl)
add_executable(test_position_sampling src/test_position_sampling.cpp)
target_link_libraries(test_position_sampling PRIVATE rigid_math)

add_executable(test_wall_constraint src/test_wall_constraint.cpp)
target_compile_options(test_wall_constraint PRIVATE -UNDEBUG)
target_link_libraries(test_wall_constraint PRIVATE rigid_math)
